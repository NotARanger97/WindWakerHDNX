cmake_minimum_required(VERSION 3.20)
project(vulkan_shader_warmup_test CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CEMU "${ROOT}/runtime/third_party/cemu")
find_package(Threads REQUIRED)
find_package(glslang CONFIG REQUIRED)
file(GLOB_RECURSE CEMU_SOURCES "${CEMU}/*.cpp")
list(FILTER CEMU_SOURCES EXCLUDE REGEX "LatteDecompilerEmitMSL[^/]*\\.cpp$")
add_library(warmup_cemu STATIC ${CEMU_SOURCES})
target_include_directories(warmup_cemu PUBLIC "${ROOT}/runtime/src" "${ROOT}/runtime/include"
    "${CEMU}" "${CEMU}/Cafe" "${ROOT}/runtime/third_party/fmt/include")
target_compile_definitions(warmup_cemu PUBLIC ENABLE_VULKAN=1)
target_compile_options(warmup_cemu PUBLIC -include "${CEMU}/cemu_shim.h" -ffunction-sections -fdata-sections)
target_compile_options(warmup_cemu PRIVATE -w)
foreach(mode desktop switch_branches)
    add_executable(warmup_${mode} "${ROOT}/runtime/tools/vulkan_shader_warmup_test.cpp")
    target_link_libraries(warmup_${mode} PRIVATE warmup_cemu glslang::glslang glslang::SPIRV
        glslang::glslang-default-resource-limits Threads::Threads)
    if(APPLE)
        target_link_options(warmup_${mode} PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(warmup_${mode} PRIVATE -Wl,--gc-sections)
    endif()
    if(mode STREQUAL switch_branches)
        target_compile_definitions(warmup_${mode} PRIVATE __SWITCH__=1)
    endif()
    foreach(enabled 1 0)
        set(cache "${CMAKE_BINARY_DIR}/cache_${mode}_${enabled}")
        add_test(NAME ${mode}_${enabled} COMMAND warmup_${mode} "${cache}")
        set_tests_properties(${mode}_${enabled} PROPERTIES
            ENVIRONMENT "WWHD_VK_SHADER_CACHE=${cache};WWHD_VK_SHADER_WARMUP=${enabled}")
    endforeach()
endforeach()
